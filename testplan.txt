    def __init__(self, output_dir: str = "reports"):
        os.makedirs(output_dir, exist_ok=True)
        self.output_dir = output_dir
        self.pdf1_name = ""
        self.pdf2_name = ""

    def _escape(self, text: Optional[Any]) -> str:
        if text is None:
            return ""
        return str(text).replace("&", "&amp;").replace("<", "&lt;").replace(">", "&gt;")

    def _calculate_text_similarity_local(self, text1: Optional[str], text2: Optional[str]) -> float:
        if text1 is None and text2 is None: return 1.0
        if text1 is None or text2 is None: return 0.0
        norm_text1 = str(text1).strip()
        norm_text2 = str(text2).strip()
        if not norm_text1 and not norm_text2: return 1.0
        if not norm_text1 or not norm_text2: return 0.0
        return SequenceMatcher(None, norm_text1, norm_text2).ratio()

    def _highlight_word_diff(self, text1: str, text2: str) -> Tuple[str, str]:
        words1 = str(text1).split()
        words2 = str(text2).split()
        matcher = SequenceMatcher(None, words1, words2)
        
        highlighted_text1_segments, highlighted_text2_segments = [], []

        for tag, i1, i2, j1, j2 in matcher.get_opcodes():
            seg1, seg2 = " ".join(words1[i1:i2]), " ".join(words2[j1:j2])
            if tag == 'equal':
                highlighted_text1_segments.append(self._escape(seg1))
                highlighted_text2_segments.append(self._escape(seg2))
            elif tag == 'delete':
                highlighted_text1_segments.append(f'<span class="diff-word-deleted">{self._escape(seg1)}</span>')
            elif tag == 'insert':
                highlighted_text2_segments.append(f'<span class="diff-word-inserted">{self._escape(seg2)}</span>')
            elif tag == 'replace':
                highlighted_text1_segments.append(f'<span class="diff-word-deleted">{self._escape(seg1)}</span>')
                highlighted_text2_segments.append(f'<span class="diff-word-inserted">{self._escape(seg2)}</span>')
        
        return " ".join(highlighted_text1_segments), " ".join(highlighted_text2_segments)

    def _generate_paired_table_diff_html(self,
                                         table1_content: Optional[List[List[str]]],
                                         table2_content: Optional[List[List[str]]],
                                         table1_id: Optional[str],
                                         table2_id: Optional[str]) -> str:
        
        def render_one_side(main_content, compare_content, is_left):
            if main_content is None: return "<p>Table not present.</p>"
            if not main_content: return "<p>Table is empty.</p>"
            
            html = ['<table class="diff-table" border="1">']
            for r_idx, row in enumerate(main_content):
                html.append("<tr>")
                for c_idx, cell in enumerate(row):
                    cell_class = "cell-similar"
                    if compare_content:
                        try:
                            compare_cell = compare_content[r_idx][c_idx]
                            if self._normalize_cell_for_render(cell) != self._normalize_cell_for_render(compare_cell):
                                cell_class = "cell-modified"
                        except IndexError:
                             cell_class = "cell-deleted" if is_left else "cell-inserted"
                    else:
                        cell_class = "cell-deleted" if is_left else "cell-inserted"
                    html.append(f'<td class="{cell_class}">{self._escape(cell)}</td>')
                html.append("</tr>")
            html.append("</table>")
            return "".join(html)

        side1_html = render_one_side(table1_content, table2_content, is_left=True)
        side2_html = render_one_side(table2_content, table1_content, is_left=False)

        html = ['<div class="table-diff-container">']
        html.append('<div class="table-diff-side">')
        html.append(f'<h6>{self._escape(table1_id or "Table")} ({self._escape(self.pdf1_name)})</h6>')
        html.append(side1_html)
        html.append('</div>')
        html.append('<div class="table-diff-separator"></div>')
        html.append('<div class="table-diff-side">')
        html.append(f'<h6>{self._escape(table2_id or "Table")} ({self._escape(self.pdf2_name)})</h6>')
        html.append(side2_html)
        html.append('</div>')
        html.append('</div>')
        return "".join(html)
        
    def _normalize_cell_for_render(self, cell_text: Any) -> str:
        return re.sub(r'\s+', ' ', str(cell_text)).strip()

    def _add_item_to_page(self, pages_dict, item, page_num):
        if page_num not in pages_dict:
            pages_dict[page_num] = {"text_differences": [], "table_differences": []}
        
        is_table = 'table1_content' in item or 'table2_content' in item
        target_list = pages_dict[page_num]["table_differences"] if is_table else pages_dict[page_num]["text_differences"]
        
        if not any(self._get_unique_item_id(i) == self._get_unique_item_id(item) for i in target_list):
            target_list.append(item)

    def _get_unique_item_id(self, item: Dict) -> Tuple:
        is_table = 'table1_content' in item or 'table2_content' in item
        item_type = "table" if is_table else "text"
        status = item["status"]

        if status in ["modified", "moved", "matched"]:
            id1 = item.get("id1") or item.get("table1_id")
            id2 = item.get("id2") or item.get("table2_id")
            uid_part1 = str(id1) if id1 else f"p{item.get('page1')}_{item.get('block1_bbox', item.get('table1_bbox'))}"
            uid_part2 = str(id2) if id2 else f"p{item.get('page2')}_{item.get('block2_bbox', item.get('table2_bbox'))}"
            return (item_type, status, tuple(sorted((uid_part1, uid_part2))))
        
        elif status == "deleted":
            id1 = item.get("id1") or item.get("table1_id")
            uid_part1 = str(id1) if id1 else f"p{item.get('page1')}_{item.get('block1_bbox', item.get('table1_bbox'))}"
            return (item_type, status, uid_part1)
        elif status == "inserted":
            id2 = item.get("id2") or item.get("table2_id")
            uid_part2 = str(id2) if id2 else f"p{item.get('page2')}_{item.get('block2_bbox', item.get('table2_bbox'))}"
            return (item_type, status, uid_part2)
        
        return (item_type, status, item.get("id1"), item.get("id2"))

    def _collect_all_renderable_items(self, results: Dict) -> Tuple[List[Dict], set]:
        all_raw_text_diffs = []
        all_raw_table_diffs = []

        for page_data in results.get("pages", {}).values():
            all_raw_text_diffs.extend(page_data.get("text_differences", []))
            all_raw_table_diffs.extend(page_data.get("table_differences", []))

        prepared_text_items = self._prepare_diffs_for_render(all_raw_text_diffs, is_table=False)
        prepared_table_items = self._prepare_diffs_for_render(all_raw_table_diffs, is_table=True)
        
        all_items_before_dedup = prepared_text_items + prepared_table_items
        
        final_unique_items = []
        processed_uids = set()
        for item in all_items_before_dedup:
            uid = self._get_unique_item_id(item)
            if uid not in processed_uids:
                final_unique_items.append(item)
                processed_uids.add(uid)
        
        return final_unique_items, processed_uids

    def _pair_deleted_and_inserted_items(self, deleted_items: List[Dict], inserted_items: List[Dict], similarity_calculator: Callable) -> Tuple[List[Dict], List[Dict], List[Dict]]:
        pairs = []
        for d_idx, d_item in enumerate(deleted_items):
            for i_idx, i_item in enumerate(inserted_items):
                similarity = similarity_calculator(d_item, i_item)
                if similarity >= self.SIMILARITY_THRESHOLD_FOR_PAIRING:
                    pairs.append({'d_idx': d_idx, 'i_idx': i_idx, 'sim': similarity})

        pairs.sort(key=lambda x: x['sim'], reverse=True)

        paired_d_indices = set()
        paired_i_indices = set()
        paired_items = []

        for pair in pairs:
            d_idx, i_idx = pair['d_idx'], pair['i_idx']
            if d_idx not in paired_d_indices and i_idx not in paired_i_indices:
                d_item = deleted_items[d_idx]
                i_item = inserted_items[i_idx]
                
                is_table = 'table1_content' in d_item
                content_key1 = 'table1_content' if is_table else 'text1'
                content_key2 = 'table2_content' if is_table else 'text2'

                page1 = d_item.get("page1")
                page2 = i_item.get("page2")
                new_status = "modified" if page1 == page2 else "moved"

                combined_item = {**d_item, **i_item, "status": new_status, "score": pair['sim']}
                combined_item[content_key1] = d_item.get(content_key1)
                combined_item[content_key2] = i_item.get(content_key2)
                
                paired_items.append(combined_item)
                paired_d_indices.add(d_idx)
                paired_i_indices.add(i_idx)
        
        unpaired_deleted = [item for i, item in enumerate(deleted_items) if i not in paired_d_indices]
        unpaired_inserted = [item for i, item in enumerate(inserted_items) if i not in paired_i_indices]
        
        return paired_items, unpaired_deleted, unpaired_inserted

    def _prepare_diffs_for_render(self, raw_diffs: List[Dict], is_table: bool) -> List[Dict]:
        
        def text_similarity(item1, item2):
            return self._calculate_text_similarity_local(item1.get("text1"), item2.get("text2"))

        def table_similarity(item1, item2):
            content1 = item1.get("table1_content")
            content2 = item2.get("table2_content")
            if not content1 or not content2: return 0.0
            rows1 = [" ".join(map(self._normalize_cell_for_render, r)) for r in content1 if r]
            rows2 = [" ".join(map(self._normalize_cell_for_render, r)) for r in content2 if r]
            if not rows1 or not rows2: return 0.0
            return SequenceMatcher(None, rows1, rows2).ratio()

        similarity_calculator = table_similarity if is_table else text_similarity

        other_items = [d for d in raw_diffs if d["status"] not in ["deleted", "inserted"]]
        deleted_items = [d for d in raw_diffs if d["status"] == "deleted"]
        inserted_items = [d for d in raw_diffs if d["status"] == "inserted"]

        paired, unpaired_deleted, unpaired_inserted = self._pair_deleted_and_inserted_items(
            deleted_items, inserted_items, similarity_calculator
        )
        
        all_items = other_items + paired + unpaired_deleted + unpaired_inserted
        final_list = []

        confidential_pattern = re.compile(r"Confidential\.*\s+\w+\s+\d{4}\s+\|\s+.*", re.IGNORECASE)

        for item in all_items:
            status = item["status"]
            score = item.get("score", 0.0)

            if (is_table and status == "modified" and score >= self.HIDE_MODIFIED_TABLE_SIMILARITY_THRESHOLD) or \
               (is_table and status == "matched" and score >= self.HIDE_MATCHED_TABLE_SIMILARITY_THRESHOLD) or \
               (status == "moved" and score == 1.0):
                continue

            if not is_table and status == "moved":
                text1 = item.get("text1", "")
                if confidential_pattern.match(str(text1).strip()):
                    continue

            display_item = item.copy()

            if is_table:
                display_item["diff_html"] = self._generate_paired_table_diff_html(
                    item.get("table1_content"), item.get("table2_content"),
                    item.get("table1_id"), item.get("table2_id")
                )
            else:
                text1, text2 = str(item.get("text1", "")), str(item.get("text2", ""))
                if status in ["modified", "moved"]:
                    h1, h2 = self._highlight_word_diff(text1, text2)
                    display_item["display_text1"], display_item["display_text2"] = h1, h2
                elif status == "deleted":
                    display_item["display_text1"] = self._escape(text1)
                    display_item["display_text2"] = f"<i>Not in {self._escape(self.pdf2_name)}</i>"
                elif status == "inserted":
                    display_item["display_text1"] = f"<i>Not in {self._escape(self.pdf1_name)}</i>"
                    display_item["display_text2"] = self._escape(text2)
                else:
                    display_item["display_text1"], display_item["display_text2"] = self._escape(text1), self._escape(text2)

            if status == "moved":
                 display_item["move_info"] = f"Moved: p.{item.get('page1')} ➜ p.{item.get('page2')} (Sim: {score*100:.1f}%)"

            final_list.append(display_item)
        
        return final_list

    def _calculate_summary(self, results: Dict, all_renderable_items: Optional[List[Dict]] = None) -> Dict:
        if all_renderable_items is None:
            all_renderable_items, _ = self._collect_all_renderable_items(results)

        s_modifications, s_insertions, s_deletions, s_moved = 0, 0, 0, 0
        text_diff_count, table_diff_count = 0, 0
        pages_with_diffs_set = set()

        for item in all_renderable_items:
            is_table = 'table1_content' in item or 'table2_content' in item
            
            if is_table: table_diff_count += 1
            else: text_diff_count += 1
            
            if item.get("page1"): pages_with_diffs_set.add(item.get("page1"))
            if item.get("page2"): pages_with_diffs_set.add(item.get("page2"))

            status = item["status"]
            if status == "modified": s_modifications += 1
            elif status == "inserted": s_insertions += 1
            elif status == "deleted": s_deletions += 1
            elif status == "moved": s_moved += 1
        
        return {
            "total_pages": results.get("max_pages", 0),
            "pages_with_differences": len(pages_with_diffs_set),
            "text_differences": text_diff_count,
            "table_differences": table_diff_count,
            "ds_modifications": s_modifications,
            "ds_insertions": s_insertions,
            "ds_deletions": s_deletions,
            "ds_moved_elements": s_moved,
        }

document.addEventListener('DOMContentLoaded', function () {
    const pageButtons = document.querySelectorAll('.page-button');
    const pageSections = document.querySelectorAll('.page-section');
    let currentActiveButton = null;
    let isClickScroll = false; 

    function setActiveButton(buttonToActivate) {
        if (currentActiveButton) currentActiveButton.classList.remove('active');
        if (buttonToActivate) {
            buttonToActivate.classList.add('active');
            currentActiveButton = buttonToActivate;
        } else {
            currentActiveButton = null;
        }
    }
    if (window.location.hash) {
        const targetButton = document.querySelector(`.page-button[href="${window.location.hash}"]`);
        setActiveButton(targetButton);
    } else if (pageButtons.length > 0) {
        setActiveButton(pageButtons[0]);
    }

    if ('IntersectionObserver' in window) {
        const observer = new IntersectionObserver((entries) => {
            if (isClickScroll) return;
            entries.forEach(entry => {
                if (entry.isIntersecting) {
                    const targetButton = document.querySelector(`.page-button[href="#${entry.target.id}"]`);
                    setActiveButton(targetButton);
                }
            });
        }, { rootMargin: '0px 0px -70% 0px' });
        pageSections.forEach(section => observer.observe(section));
    }
    pageButtons.forEach(button => {
        button.addEventListener('click', function(e) {
            e.preventDefault();
            isClickScroll = true;
            const targetId = this.getAttribute('href');
            const targetSection = document.querySelector(targetId);
            setActiveButton(this); 
            if (targetSection) {
                const headerOffset = document.querySelector('.page-nav').offsetHeight + 15;
                const elementPosition = targetSection.getBoundingClientRect().top;
                const offsetPosition = elementPosition + window.pageYOffset - headerOffset;
                window.scrollTo({ top: offsetPosition, behavior: "smooth" });
                if(history.pushState) history.pushState(null, null, targetId);
                else window.location.hash = targetId;
            }
            setTimeout(() => { isClickScroll = false; }, 1000); 
        });
    });
